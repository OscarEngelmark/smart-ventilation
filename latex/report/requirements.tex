\defreq{FR-1}
  {Functional}
  {CO₂ readings and head counts are saved in the History store}
  {Must}
  {A full simulated weekday has a stored CO₂ reading for every 10~s and a head
   count for every room minute}
  {Comparison week}

\defreq{FR-2}
  {Functional}
  {The Decision service sets the damper in BuildSim from the CO₂ readings, in
   time to hold the threshold}
  {Must}
  {After a reading above the 950 ppm target
   (Section~\ref{sec:decision-logic}), a command to open the damper fully is
   stored, and BuildSim holds the damper fully open within 2 min of room time
   of the reading}
  {End-to-end test}

\defreq{FR-3}
  {Functional}
  {A damper change shows in the CO₂ readings that follow}
  {Must}
  {After the damper opens, the third CO₂ reading is lower than the first}
  {End-to-end test}

\defreq{FR-4}
  {Functional}
  {Other components can fetch a room's stored readings and commands from a
   given time on}
  {Must}
  {The Storage service answers requests over REST, and the History
   store's query for one room from a given time returns exactly that room's
   rows from that time on, oldest first; commands are led by the one in force
   at that time}
  {Unit tests, \texttt{internal/store}; end-to-end test}

\defreq{FR-5}
  {Functional}
  {The Room-model service's fit learns how the room's CO₂ responds to people
   and to the damper, from a history of readings, head counts and damper
   levels}
  {Must}
  {For a simulated room, the fit recovers the room's actual rates (Section~\ref{sec:decision-logic}). It gives no room
   model when the history can't tell these apart}
  {Unit tests, \texttt{internal/roommodel}}

\defreq{FR-6}
  {Functional}
  {Without a room model, the Decision service's choice of level follows CO₂
   alone}
  {Must}
  {With no room model, the chosen level is fully open at 950 ppm, closed
   below 800 ppm, and unchanged in between}
  {Unit tests, \texttt{internal/planner} and \texttt{cmd/decision}}

\defreq{FR-7}
  {Functional}
  {The Dashboard shows CO₂, head count and damper level}
  {Should}
  {All three are shown over recent room time}
  {Screenshot (Section~\ref{sec:dashboard})}

\defreq{FR-8}
  {Functional}
  {The Storage service keeps bad messages out of the History store}
  {Must}
  {A message that fails its schema, a reading for another room than its topic
   names, and a second copy of a stored reading are not stored}
  {Unit tests, \texttt{cmd/\allowbreak storage-service} and
   \texttt{internal/store}}

\defreq{FR-9}
  {Functional}
  {The Occupancy simulator's schedule fills the room on weekdays}
  {Must}
  {Nobody at night or at weekends, 3 people mid-morning and mid-afternoon,
   empty at 12:00, full (6) at 13:30, never over capacity}
  {Unit tests, \texttt{internal/\allowbreak occupancy}}

\defreq{FR-10}
  {Functional}
  {In the Physical model's mass balance, CO₂ follows the people and the
   damper}
  {Must}
  {With the damper closed, a full room settles at about 3640 ppm and one
   person at about 957 ppm; fully open, a full room settles at about 903 ppm;
   an empty room returns to the outdoor 420 ppm}
  {Unit tests, \texttt{internal/co2}}

\defreq{REG-1}
  {Regulatory}
  {CO₂ stays under 1000 ppm}
  {Must}
  {The peak CO₂ over a full simulated weekday, meeting included, is under 1000
   ppm}
  {Comparison week}

\defreq{NFR-1}
  {Non-functional}
  {A killed container recovers without help}
  {Must}
  {Docker restarts it, a sensor or the Damper actuator registers its device
   with BuildSim again, and after a reading above the target, BuildSim holds
   the damper fully open within 2 min of the kill, with room time running as
   fast as the wall clock}
  {Crash test, \texttt{test/crash/crash.sh}}

\defreq{NFR-2}
  {Non-functional}
  {The ventilation runs as low as it can}
  {Should}
  {The time-weighted mean damper level over a room day is lower than under the
   CO₂-only switch of \req{FR-6}}
  {Comparison week}

\defreq{NFR-3}
  {Non-functional}
  {A faulty CO₂ sensor or a stopped Message broker does not let CO₂ through the
   threshold}
  {Should}
  {With the CO₂ sensor frozen at one value, or the Message broker stopped, CO₂
   stays under 1000 ppm}
  {Fault tests}

\defreq{NFR-4}
  {Non-functional}
  {A command follows its reading in time to hold the threshold}
  {Could}
  {The time from a reading to its command stays under 2 min of room time}
  {Delay measurement}
