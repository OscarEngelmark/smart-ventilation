% State diagram of the Decision service (cmd/decision/main.go). Every
% transition is a path in that file, internal/mqttclient or
% internal/shutdown; Docker's part is restart: on-failure in
% docker-compose.yml. Included by latex/report/report.tex.
\begin{tikzpicture}[
    font=\footnotesize, >=stealth,
    state/.style={draw, rounded corners, align=center, minimum height=0.9cm, minimum width=2.4cm},
    lbl/.style={font=\scriptsize, align=center, fill=white, inner sep=1pt},
    dot/.style={circle, fill, inner sep=2.5pt},
    every loop/.style={->, looseness=5},
  ]
  \node[dot] (start) at (0, 0) {};
  \node[state] (conn) at (2.8, 0) {Connecting};
  \node[state] (sw) at (7.6, 0) {On CO₂ alone};
  \node[state] (plan) at (13.2, 0) {Planning};
  \node[state, minimum width=7.6cm] (wait) at (10.4, -3) {Waiting for the broker};
  \node[state] (stop) at (13.2, -5.3) {Stopped};
  \node[dot, double, double distance=1.5pt] (end) at (15.4, -5.3) {};

  \draw[dashed] (0.7, 2.3) rectangle (15.6, -4.1);
  \node[anchor=south west] at (0.7, -4.1) {Running};

  \draw[->] (start) -- (conn);
  \draw (conn) edge[loop above] node[lbl, above] {connect or subscribe fails:\\exit 1, Docker restarts} (conn);
  \draw[->] (conn) -- node[lbl, above] {subscribed} (sw);
  \draw (sw) edge[loop above] node[lbl, above] {CO₂ reading:\\open or close} (sw);
  \draw[->] (sw) -- node[lbl, above] {room model arrives} (plan);
  \draw (plan) edge[loop above] node[lbl, above] {CO₂ reading: plan;\\not applied: retry\\on the next reading} (plan);

  \draw[->] ([xshift=-4mm]sw.south) -- node[lbl, left] {connection lost} ([xshift=-4mm]sw.south |- wait.north);
  \draw[<-] ([xshift=4mm]sw.south) -- node[lbl, right] {reconnected,\\no room model} ([xshift=4mm]sw.south |- wait.north);
  \draw[->] ([xshift=-4mm]plan.south) -- node[lbl, left] {connection lost} ([xshift=-4mm]plan.south |- wait.north);
  \draw[<-] ([xshift=4mm]plan.south) -- node[lbl, right] {reconnected,\\room model held} ([xshift=4mm]plan.south |- wait.north);

  \draw[->] (stop |- 0,-4.1) -- node[lbl, left] {Docker stop: exit 0} (stop);
  \draw[->] (stop) -- (end);
\end{tikzpicture}
